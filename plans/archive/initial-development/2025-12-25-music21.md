Create script to analyze all Bach chorales and extract chord progression patterns:

1. Parse Bach chorales using `music21.corpus.parse('bach/bwv*.xml')`
2. For each chorale:
   - Use `.chordify()` to extract vertical harmonies
   - Apply `roman.romanNumeralFromChord()` to get scale degree analysis
   - Track all (chord_i → chord_j) transitions
   - Identify cadence patterns (last 3-4 chords before endings)
3. Build transition probability matrix: `transitions[degree_i][degree_j] = probability`
4. Extract common cadence approaches for each type (Perfect, Plagal, etc.)

```python
class ChordProgressionGenerator:
    """Generates chord progressions using music21 and Bach corpus patterns."""

    def __init__(self,
                 min_length: int = 4,
                 max_length: int = 8,
                 use_voice_leading: bool = True,
                 use_sevenths: bool = True,
                 use_corpus: bool = True,
                 corpus_temperature: float = 0.8,
                 key: str = 'C'):  # NEW: support multiple keys
        self.min_length = min_length
        self.max_length = max_length
        self.use_voice_leading = use_voice_leading
        self.use_sevenths = use_sevenths
        self.use_corpus = use_corpus
        self.key = key

        self.voice_leader = VoiceLeader() if use_voice_leading else None
        self.markov_selector = MarkovChordSelector() if use_corpus else None
        self.corpus_temperature = corpus_temperature

    def generate_progression(self, cadence_type: CadenceType) -> List[roman.RomanNumeral]:
        """
        Generate chord progression ending with specified cadence.

        Changes from old version:
        - Returns music21 RomanNumeral objects (not custom Chord)
        - Uses Markov model for chord selection (not hardcoded rules)
        - Supports any key (not just C major)
        """
        length = random.randint(self.min_length, self.max_length)

        # Get cadence pattern (e.g., V→I)
        cadence_degrees = CadencePattern.get_cadence_chords(cadence_type)

        # Generate intro leading to cadence
        intro_length = length - 2
        intro_degrees = self._generate_intro(intro_length, cadence_degrees[0], cadence_type)

        # Combine intro + cadence
        all_degrees = intro_degrees + list(cadence_degrees)

        # Convert to RomanNumeral objects
        progression = []
        for i, degree in enumerate(all_degrees):
            # Use sevenths on degrees 2, 5, 6, 7 (except first chord)
            use_seventh = (self.use_sevenths and
                          degree in [2, 5, 6, 7] and
                          i > 0)

            chord = ChordFactory.create_chord(degree, self.key, use_seventh)
            progression.append(chord)

        return progression

    def _generate_intro(self, length: int, target_degree: int,
                       cadence_type: CadenceType) -> List[int]:
        """Generate intro chords using Markov model or rule fallback."""
        if length == 0:
            return []

        # Always start with I (tonic)
        intro = [1]
        current = 1

        # Generate middle chords
        for i in range(length - 1):
            if self.use_corpus and self.markov_selector:
                # Use Bach corpus patterns
                next_degree = self.markov_selector.get_next_chord(
                    current, cadence_type, self.corpus_temperature
                )
                if next_degree is None:
                    # Corpus had no good option, use rule fallback
                    next_degree = self._choose_next_chord_rules(current, cadence_type)
            else:
                # Rule-based selection
                next_degree = self._choose_next_chord_rules(current, cadence_type)

            intro.append(next_degree)
            current = next_degree

        return intro

    def _choose_next_chord_rules(self, current: int, cadence_type: CadenceType) -> int:
        """
        Fallback rule-based selection (simplified from old version).

        Strong progressions:
        - I → IV, V, vi
        - ii → V
        - IV → V, I
        - V → I, vi
        - vi → IV, ii
        """
        STRONG_NEXT = {
            1: [4, 5, 6],
            2: [5],
            3: [4, 6],
            4: [5, 1],
            5: [1, 6],
            6: [4, 2],
            7: [1]
        }

        valid = STRONG_NEXT.get(current, [1, 5])
        return random.choice(valid)

    def progression_to_midi(self, progression: List[roman.RomanNumeral]) -> List[List[int]]:
        """Convert progression to MIDI notes with voice leading."""
        if self.use_voice_leading and self.voice_leader:
            return self.voice_leader.voice_progression(progression)
        else:
            # Simple root position fallback
            return [ChordFactory.get_midi_notes(chord, None)
                    for chord in progression]

    def progression_to_symbols(self, progression: List[roman.RomanNumeral]) -> List[str]:
        """Convert progression to Roman numeral strings."""
        return [chord.figure for chord in progression]
```
